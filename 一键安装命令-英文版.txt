# ========================================
# Miner Manager Tool - One-Click Install Command
# ========================================
# Usage: Copy the command below, paste it into CentOS Stream 9 terminal and press Enter
# ========================================

cat > miner-manager.sh << 'EOF'
#!/bin/bash

# ========================================
# Miner Manager Script - CentOS Stream 9
# Functions: Download, Install, Start, Manage Miner
# ========================================

# Define colors for better UI
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

# Default configuration
DEFAULT_WORKER_NAME="qiqi-1"
MINER_DIR="/root/miner"
MINER_BINARY="xmrig"
PID_FILE="/tmp/miner.pid"
LOG_FILE="/tmp/miner.log"

# ========================================
# Function: Show Main Menu
# Description: Main interface for user to select operations
# ========================================
show_menu() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Miner Manager Tool v1.0${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    echo -e "${GREEN}1.${NC} Download and Install Miner"
    echo -e "${GREEN}2.${NC} Set Worker Name (Current: ${YELLOW}${WORKER_NAME}${NC})"
    echo -e "${GREEN}3.${NC} Start Miner (Foreground)"
    echo -e "${GREEN}4.${NC} Start Miner (Background)"
    echo -e "${GREEN}5.${NC} Check Status"
    echo -e "${GREEN}6.${NC} Stop Miner"
    echo -e "${GREEN}7.${NC} View Logs"
    echo -e "${GREEN}0.${NC} Exit"
    echo ""
    echo -e "${CYAN}========================================${NC}"
    echo -n -e "${YELLOW}Please select [0-7]: ${NC}"
}

# ========================================
# Function: Download and Install Miner
# Description: Download from URL, extract, delete archive, set permissions
# ========================================
download_miner() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Download and Install Miner${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Prompt for download URL
    echo -n "Enter download URL (press Enter for default): "
    read DOWNLOAD_URL
    
    # Use default URL if not provided
    if [ -z "$DOWNLOAD_URL" ]; then
        DOWNLOAD_URL="https://github.com/xmrig/xmrig/releases/download/v6.26.0/xmrig-6.26.0-linux-static-x64.tar.gz"
        echo -e "${YELLOW}Using default download URL${NC}"
    fi
    
    # Create miner directory
    echo -e "${BLUE}[1/5]${NC} Creating installation directory..."
    mkdir -p "$MINER_DIR"
    
    # Enter directory
    cd "$MINER_DIR" || {
        echo -e "${RED}Error: Cannot enter directory $MINER_DIR${NC}"
        read -p "Press Enter to return to menu..."
        return 1
    }
    
    # Get filename from URL
    FILENAME=$(basename "$DOWNLOAD_URL")
    
    # Check if file already exists
    if [ -f "$FILENAME" ]; then
        echo -e "${YELLOW}File exists. Re-download? (y/n)${NC}"
        read -r RE_DOWNLOAD
        if [ "$RE_DOWNLOAD" != "y" ] && [ "$RE_DOWNLOAD" != "Y" ]; then
            echo -e "${YELLOW}Skipping download${NC}"
        else
            rm -f "$FILENAME"
        fi
    fi
    
    # Download file
    if [ ! -f "$FILENAME" ]; then
        echo -e "${BLUE}[2/5]${NC} Downloading: $DOWNLOAD_URL"
        echo -e "${YELLOW}This may take a few minutes, please wait...${NC}"
        
        # Try wget first, then curl
        if command -v wget &> /dev/null; then
            wget --show-progress -O "$FILENAME" "$DOWNLOAD_URL"
        elif command -v curl &> /dev/null; then
            curl -L -o "$FILENAME" "$DOWNLOAD_URL"
        else
            echo -e "${RED}Error: wget or curl not installed${NC}"
            echo -e "${YELLOW}Please install: dnf install -y wget${NC}"
            read -p "Press Enter to return to menu..."
            return 1
        fi
        
        # Check if download succeeded
        if [ $? -ne 0 ]; then
            echo -e "${RED}Error: Download failed!${NC}"
            read -p "Press Enter to return to menu..."
            return 1
        fi
        
        echo -e "${GREEN}Download complete${NC}"
    fi
    
    # Extract file
    echo -e "${BLUE}[3/5]${NC} Extracting..."
    tar -xzf "$FILENAME"
    
    if [ $? -ne 0 ]; then
        echo -e "${RED}Error: Extraction failed!${NC}"
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    echo -e "${GREEN}Extraction complete${NC}"
    
    # Delete archive
    echo -e "${BLUE}[4/5]${NC} Deleting archive..."
    rm -f "$FILENAME"
    echo -e "${GREEN}Archive deleted${NC}"
    
    # Find xmrig and set permissions
    echo -e "${BLUE}[5/5]${NC} Setting file permissions..."
    
    # Look for xmrig executable
    if [ -f "$MINER_BINARY" ]; then
        chmod +x "$MINER_BINARY"
        echo -e "${GREEN}Permissions set${NC}"
    else
        # Try to find in subdirectories
        FOUND=0
        for dir in */; do
            if [ -f "$dir$MINER_BINARY" ]; then
                chmod +x "$dir$MINER_BINARY"
                # Create symlink to main directory
                ln -sf "$dir$MINER_BINARY" "$MINER_BINARY"
                FOUND=1
                echo -e "${GREEN}Permissions set${NC}"
                break
            fi
        done
        
        if [ $FOUND -eq 0 ]; then
            echo -e "${RED}Warning: $MINER_BINARY not found${NC}"
            echo -e "${YELLOW}Current directory contents:${NC}"
            ls -la
        fi
    fi
    
    echo ""
    echo -e "${GREEN}========================================${NC}"
    echo -e "${GREEN}       Installation Complete!${NC}"
    echo -e "${GREEN}========================================${NC}"
    echo -e "${YELLOW}Install directory: $MINER_DIR${NC}"
    echo -e "${YELLOW}Executable: $MINER_DIR/$MINER_BINARY${NC}"
    echo ""
    
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Set Worker Name
# Description: Set custom worker name, default is qiqi-1
# ========================================
set_worker_name() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Set Worker Name${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    echo -e "${YELLOW}Current worker name: ${WORKER_NAME}${NC}"
    echo ""
    echo -n "Enter new worker name (press Enter to keep): "
    read NEW_NAME
    
    # If user entered new name
    if [ -n "$NEW_NAME" ]; then
        WORKER_NAME="$NEW_NAME"
        # Save to config file
        echo "WORKER_NAME=$WORKER_NAME" > "$MINER_DIR/miner.conf"
        echo -e "${GREEN}Worker name updated to: ${WORKER_NAME}${NC}"
    else
        echo -e "${YELLOW}Worker name unchanged${NC}"
    fi
    
    echo ""
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Start Miner (Foreground)
# Description: Start miner in foreground with real-time output
# ========================================
start_miner() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Start Miner (Foreground)${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Check if miner binary exists
    if [ ! -f "$MINER_DIR/$MINER_BINARY" ]; then
        echo -e "${RED}Error: Miner binary not found!${NC}"
        echo -e "${YELLOW}Please run option 1 to download and install${NC}"
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    # Check if already running
    if [ -f "$PID_FILE" ]; then
        PID=$(cat "$PID_FILE")
        if ps -p "$PID" > /dev/null 2>&1; then
            echo -e "${YELLOW}Warning: Miner already running (PID: $PID)${NC}"
            echo -n "Stop and restart? (y/n): "
            read RESTART
            if [ "$RESTART" = "y" ] || [ "$RESTART" = "Y" ]; then
                stop_miner_silent
            else
                read -p "Press Enter to return to menu..."
                return 1
            fi
        fi
    fi
    
    echo -e "${BLUE}Worker name: ${WORKER_NAME}${NC}"
    echo -e "${BLUE}Start command:${NC}"
    echo -e "${YELLOW}cd $MINER_DIR && ./$MINER_BINARY -a gr -o stratum+ssl://ghostrider.unmineable.com:443 -u USDT:TDFRoYVFwze54ojydkiPUja8Twix9X5QRR.${WORKER_NAME}#ek8v-txze -p x${NC}"
    echo ""
    echo -e "${YELLOW}Press Ctrl+C to stop miner${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Start miner
    cd "$MINER_DIR" || return 1
    ./$MINER_BINARY -a gr -o stratum+ssl://ghostrider.unmineable.com:443 -u "USDT:TDFRoYVFwze54ojydkiPUja8Twix9X5QRR.${WORKER_NAME}#ek8v-txze" -p x
    
    echo ""
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Start Miner (Background)
# Description: Start miner in background using nohup
# ========================================
start_miner_background() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Start Miner (Background)${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Check if miner binary exists
    if [ ! -f "$MINER_DIR/$MINER_BINARY" ]; then
        echo -e "${RED}Error: Miner binary not found!${NC}"
        echo -e "${YELLOW}Please run option 1 to download and install${NC}"
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    # Check if already running
    if [ -f "$PID_FILE" ]; then
        PID=$(cat "$PID_FILE")
        if ps -p "$PID" > /dev/null 2>&1; then
            echo -e "${YELLOW}Warning: Miner already running (PID: $PID)${NC}"
            echo -n "Stop and restart? (y/n): "
            read RESTART
            if [ "$RESTART" = "y" ] || [ "$RESTART" = "Y" ]; then
                stop_miner_silent
            else
                read -p "Press Enter to return to menu..."
                return 1
            fi
        fi
    fi
    
    echo -e "${BLUE}Starting miner in background...${NC}"
    
    # Start with nohup in background, redirect output to log
    cd "$MINER_DIR" || return 1
    nohup ./$MINER_BINARY -a gr -o stratum+ssl://ghostrider.unmineable.com:443 -u "USDT:TDFRoYVFwze54ojydkiPUja8Twix9X5QRR.${WORKER_NAME}#ek8v-txze" -p x > "$LOG_FILE" 2>&1 &
    
    # Save process ID
    echo $! > "$PID_FILE"
    
    # Wait a few seconds to check if started successfully
    sleep 3
    
    if ps -p $(cat "$PID_FILE") > /dev/null 2>&1; then
        echo -e "${GREEN}Miner started successfully in background${NC}"
        echo -e "${YELLOW}Process ID: $(cat $PID_FILE)${NC}"
        echo -e "${YELLOW}Log file: $LOG_FILE${NC}"
        echo ""
        echo -e "${CYAN}Tips:${NC}"
        echo -e "  - Use option 5 to check status"
        echo -e "  - Use option 7 to view logs"
        echo -e "  - Use option 6 to stop miner"
    else
        echo -e "${RED}Error: Failed to start miner!${NC}"
        echo -e "${YELLOW}Check log file: $LOG_FILE${NC}"
        rm -f "$PID_FILE"
    fi
    
    echo ""
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Check Status
# Description: Check if miner is running, show process info and resource usage
# ========================================
check_status() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Check Status${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Check if PID file exists
    if [ ! -f "$PID_FILE" ]; then
        echo -e "${RED}PID file not found, miner may not be running${NC}"
        echo ""
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    # Read PID
    PID=$(cat "$PID_FILE")
    
    # Check if process exists
    if ps -p "$PID" > /dev/null 2>&1; then
        echo -e "${GREEN}Miner is running${NC}"
        echo ""
        echo -e "${YELLOW}Process Info:${NC}"
        echo -e "  PID: ${GREEN}$PID${NC}"
        echo -e "  Worker Name: ${GREEN}$WORKER_NAME${NC}"
        echo -e "  Install Dir: $MINER_DIR"
        echo -e "  Log File: $LOG_FILE"
        echo ""
        echo -e "${YELLOW}Resource Usage:${NC}"
        ps -p "$PID" -o pid,ppid,%cpu,%mem,etime,cmd --no-headers | while read line; do
            echo -e "  $line"
        done
        echo ""
        echo -e "${YELLOW}Last 10 log lines:${NC}"
        if [ -f "$LOG_FILE" ]; then
            tail -n 10 "$LOG_FILE" | while read line; do
                echo -e "  $line"
            done
        else
            echo -e "  ${RED}Log file not found${NC}"
        fi
    else
        echo -e "${RED}Miner is not running${NC}"
        echo -e "${YELLOW}PID file exists but process stopped${NC}"
        echo -e "${YELLOW}Suggestion: Delete PID file and restart${NC}"
        rm -f "$PID_FILE"
    fi
    
    echo ""
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Stop Miner
# Description: Stop running miner process
# ========================================
stop_miner() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       Stop Miner${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    # Check if PID file exists
    if [ ! -f "$PID_FILE" ]; then
        echo -e "${YELLOW}PID file not found, miner may not be running${NC}"
        echo ""
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    # Read PID
    PID=$(cat "$PID_FILE")
    
    # Check if process exists
    if ps -p "$PID" > /dev/null 2>&1; then
        echo -e "${YELLOW}Stopping miner (PID: $PID)...${NC}"
        
        # Try graceful stop
        kill "$PID"
        
        # Wait for process to end
        for i in {1..10}; do
            if ! ps -p "$PID" > /dev/null 2>&1; then
                echo -e "${GREEN}Miner stopped successfully${NC}"
                rm -f "$PID_FILE"
                echo ""
                read -p "Press Enter to return to menu..."
                return 0
            fi
            sleep 1
            echo -n "."
        done
        
        # If graceful stop failed, force stop
        echo ""
        echo -e "${YELLOW}Graceful stop failed, trying force stop...${NC}"
        kill -9 "$PID"
        sleep 1
        
        if ! ps -p "$PID" > /dev/null 2>&1; then
            echo -e "${GREEN}Miner force stopped${NC}"
            rm -f "$PID_FILE"
        else
            echo -e "${RED}Error: Cannot stop miner process${NC}"
        fi
    else
        echo -e "${YELLOW}Process not found, cleaning PID file${NC}"
        rm -f "$PID_FILE"
    fi
    
    echo ""
    read -p "Press Enter to return to menu..."
}

# ========================================
# Function: Stop Miner Silently (Internal Use)
# Description: Stop miner without displaying messages
# ========================================
stop_miner_silent() {
    if [ -f "$PID_FILE" ]; then
        PID=$(cat "$PID_FILE")
        if ps -p "$PID" > /dev/null 2>&1; then
            kill "$PID"
            sleep 2
            if ps -p "$PID" > /dev/null 2>&1; then
                kill -9 "$PID"
            fi
        fi
        rm -f "$PID_FILE"
    fi
}

# ========================================
# Function: View Logs
# Description: Display miner runtime logs
# ========================================
view_logs() {
    clear
    echo -e "${CYAN}========================================${NC}"
    echo -e "${CYAN}       View Logs${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    if [ ! -f "$LOG_FILE" ]; then
        echo -e "${RED}Log file not found: $LOG_FILE${NC}"
        echo -e "${YELLOW}Miner may not have been started yet${NC}"
        echo ""
        read -p "Press Enter to return to menu..."
        return 1
    fi
    
    echo -e "${YELLOW}Log file: $LOG_FILE${NC}"
    echo -e "${YELLOW}Showing last 50 lines:${NC}"
    echo -e "${CYAN}========================================${NC}"
    echo ""
    
    tail -n 50 "$LOG_FILE"
    
    echo ""
    echo -e "${CYAN}========================================${NC}"
    echo -e "${YELLOW}Tips:${NC}"
    echo -e "  - Use 'tail -f $LOG_FILE' for real-time logs"
    echo -e "  - Use 'cat $LOG_FILE' for full logs"
    echo ""
    
    read -p "Press Enter to return to menu..."
}

# ========================================
# Main Program
# Description: Program entry point, load config and show menu
# ========================================

# Load config file
if [ -f "$MINER_DIR/miner.conf" ]; then
    source "$MINER_DIR/miner.conf"
else
    WORKER_NAME="$DEFAULT_WORKER_NAME"
fi

# Main loop
while true; do
    show_menu
    read -r CHOICE
    
    case $CHOICE in
        1)
            download_miner
            ;;
        2)
            set_worker_name
            ;;
        3)
            start_miner
            ;;
        4)
            start_miner_background
            ;;
        5)
            check_status
            ;;
        6)
            stop_miner
            ;;
        7)
            view_logs
            ;;
        0)
            clear
            echo -e "${GREEN}Thank you for using Miner Manager Tool. Goodbye!${NC}"
            echo ""
            exit 0
            ;;
        *)
            echo -e "${RED}Invalid choice, please try again${NC}"
            sleep 1
            ;;
    esac
done
EOF

chmod +x miner-manager.sh && ./miner-manager.sh
